cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" TASKBAR_ORDER_VERSION LIMIT_COUNT 1)
project(TaskbarOrder VERSION "${TASKBAR_ORDER_VERSION}" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

if(BUILD_TESTING)
  add_executable(order_tests tests/order_tests.cpp)
  target_include_directories(order_tests PRIVATE src)
  add_test(NAME ordering COMMAND order_tests)
  add_executable(accessor_tests tests/accessor_tests.cpp)
  target_include_directories(accessor_tests PRIVATE src)
  add_test(NAME accessor_layout COMMAND accessor_tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME offline_symbol_report COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_symbol_report.py")
  endif()
endif()

if(NOT WIN32)
  message(STATUS "Non-Windows host: building portable logic tests only. Use scripts/build-windows.ps1 for the application.")
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM|aarch64)" OR CMAKE_GENERATOR_PLATFORM MATCHES "ARM")
  message(FATAL_ERROR "TaskbarOrder must target 64-bit x86 Windows.")
endif()
if(NOT MSVC AND NOT MINGW)
  message(FATAL_ERROR "TaskbarOrder requires either MSVC or a MinGW-w64 x64 cross compiler.")
endif()
enable_language(RC)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/src/compat")
if(MSVC)
  add_compile_options(/utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  add_link_options(-static -static-libgcc -static-libstdc++)
endif()

set(CPPWINRT_INCLUDE_DIR "" CACHE PATH "Directory containing the generated winrt headers")
if(NOT EXISTS "${CPPWINRT_INCLUDE_DIR}/winrt/Microsoft.UI.Xaml.Controls.h")
  message(FATAL_ERROR "WinUI 2 projections are required. Use scripts/build-windows.ps1, or supply CPPWINRT_INCLUDE_DIR with both Windows SDK and Microsoft.UI.Xaml 2.8.7 projections.")
endif()
if(CPPWINRT_INCLUDE_DIR)
  include_directories("${CPPWINRT_INCLUDE_DIR}")
endif()
set(WINDOWS_SDK_UM_INCLUDE_DIR "" CACHE PATH "Optional Windows SDK um headers needed by C++/WinRT")
if(WINDOWS_SDK_UM_INCLUDE_DIR)
  include_directories("${WINDOWS_SDK_UM_INCLUDE_DIR}")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Use a statically linked MinHook" FORCE)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)

set(TASKBAR_ORDER_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${TASKBAR_ORDER_GENERATED_DIR}")
set(TBO_GENERATED_MANIFEST "${TASKBAR_ORDER_GENERATED_DIR}/TaskbarOrder.manifest")
configure_file(src/common/version.hpp.in "${TASKBAR_ORDER_GENERATED_DIR}/version.hpp" @ONLY)
configure_file(src/controller/app.manifest.in "${TBO_GENERATED_MANIFEST}" @ONLY)
configure_file(src/controller/app.rc.in "${TASKBAR_ORDER_GENERATED_DIR}/TaskbarOrder.rc" @ONLY)
configure_file(src/engine/hook.rc.in "${TASKBAR_ORDER_GENERATED_DIR}/TaskbarOrderHook.rc" @ONLY)
include_directories("${TASKBAR_ORDER_GENERATED_DIR}")

add_library(taskbar_common STATIC src/common/windows.cpp)
target_include_directories(taskbar_common PUBLIC src)
target_link_libraries(taskbar_common PUBLIC shell32 version ole32 uuid)

add_library(TaskbarOrderHook SHARED
  src/engine/bootstrap.cpp
  src/engine/reorder.cpp
  "${TASKBAR_ORDER_GENERATED_DIR}/TaskbarOrderHook.rc"
)
target_link_libraries(TaskbarOrderHook PRIVATE taskbar_common minhook comctl32 ole32 runtimeobject)
if(MSVC)
  target_link_libraries(TaskbarOrderHook PRIVATE windowsapp)
  target_compile_options(TaskbarOrderHook PRIVATE /W4 /EHsc)
endif()
set_target_properties(TaskbarOrderHook PROPERTIES OUTPUT_NAME "TaskbarOrderHook-${PROJECT_VERSION}" PREFIX "")

add_executable(TaskbarOrder WIN32
  src/controller/main.cpp
  src/controller/symbols.cpp
  "${TASKBAR_ORDER_GENERATED_DIR}/TaskbarOrder.rc"
)
target_link_libraries(TaskbarOrder PRIVATE taskbar_common dbghelp winhttp comctl32)
if(MSVC)
  target_compile_options(TaskbarOrder PRIVATE /W4 /EHsc)
  # The executable embeds its configured manifest through TaskbarOrder.rc.
  # Disable MSVC's generated manifest to avoid two RT_MANIFEST resources.
  target_link_options(TaskbarOrder PRIVATE /MANIFEST:NO)
elseif(MINGW)
  target_link_options(TaskbarOrder PRIVATE -municode)
endif()
add_dependencies(TaskbarOrder TaskbarOrderHook)
install(TARGETS TaskbarOrder TaskbarOrderHook RUNTIME DESTINATION .)
install(FILES VERSION README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES third_party/minhook/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
